## To create a new Folder and new FILE in gitHub using VS CODE in Powershell Terminal
#let check command for new FOLDER
  New-Item [ Folder Name ] -ItemType Directory 
#Let check commend for new File ( any tipe of file )
  New-Item [ FILE NAME.txt ] -ItemType File

# see all the file and folder 
  dir

## To create a new file in the folder in gitHub using VS CODE in Powershell Terminal
#let check command for cheate new FILE under a FOLDER
  New-Item [Folder Name ] \ [ File Name ].txt -ItemType File

## How to add file in git by VS_CODE
  git add "FILE NAME"
  git status
  git commit
  git commit -m "FILE NAME" folder with test files"
  git push

## Since you modified "FILE NAME" just need to stage → commit → push the change.
  git add FOLDER NAME/ FILE NAME.txt
  git status
  git commit -m "Update FILE NAME.txt"
  git push
